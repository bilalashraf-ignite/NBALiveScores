Set up Lighthouse CI for automated performance testing on every PR and deployment:

1. Install Lighthouse CI:
   ```bash
   npm install --save-dev @lhci/cli
   ```

2. Create Lighthouse CI workflow (.github/workflows/lighthouse-ci.yml):
   ```yaml
   name: Lighthouse CI
   on:
     pull_request:
       branches: [main]
     push:
       branches: [main]

   jobs:
     lighthouse:
       runs-on: ubuntu-latest
       steps:
         - uses: actions/checkout@v4
         - uses: actions/setup-node@v4
           with:
             node-version: '20'
             cache: 'npm'
         - name: Install dependencies
           run: npm ci
         - name: Build application
           run: npm run build
         - name: Run Lighthouse CI
           run: |
             npm install -g @lhci/cli
             lhci autorun
           env:
             LHCI_GITHUB_APP_TOKEN: ${{ secrets.LHCI_GITHUB_APP_TOKEN }}
   ```

3. Create Lighthouse CI config (lighthouserc.json):
   ```json
   {
     "ci": {
       "collect": {
         "staticDistDir": ".next",
         "url": [
           "http://localhost:3000/",
           "http://localhost:3000/?league=NBA"
         ],
         "numberOfRuns": 3
       },
       "assert": {
         "preset": "lighthouse:recommended",
         "assertions": {
           "categories:performance": ["error", {"minScore": 0.9}],
           "largest-contentful-paint": ["error", {"maxNumericValue": 2500}],
           "cumulative-layout-shift": ["error", {"maxNumericValue": 0.1}],
           "speed-index": ["error", {"maxNumericValue": 3000}],
           "interactive": ["error", {"maxNumericValue": 3500}],
           "first-contentful-paint": ["error", {"maxNumericValue": 1800}]
         }
       },
       "upload": {
         "target": "temporary-public-storage"
       }
     }
   }
   ```

4. Add npm script to package.json:
   ```json
   "scripts": {
     "lighthouse": "lhci autorun"
   }
   ```

Why Lighthouse CI: Industry-standard performance measurement, catches regressions automatically
Why LCP < 2.5s: Core Web Vitals "good" threshold for Largest Contentful Paint
Why CLS < 0.1: Core Web Vitals "good" threshold for Cumulative Layout Shift
Why 3 runs: Averages out network variance for stable measurements

3. Create skeleton dimensions test (tests/components/skeleton-dimensions.test.tsx):
   ```tsx
   import { render } from '@testing-library/react';
   import { GameCard } from '@/components/game-card';
   import { GameCardSkeleton } from '@/components/game-card-skeleton';

   describe('Skeleton Dimension Matching', () => {
     it('GameCardSkeleton matches GameCard minimum height', () => {
       const mockGame = {
         id: '1',
         homeTeam: { name: 'Lakers', abbreviation: 'LAL', score: 100 },
         awayTeam: { name: 'Celtics', abbreviation: 'BOS', score: 98 },
         state: 'LIVE'
       };

       const { container: skeletonContainer } = render(<GameCardSkeleton />);
       const { container: cardContainer } = render(
         <GameCard game={mockGame} onClick={() => {}} index={0} />
       );

       const skeletonRect = (skeletonContainer.firstChild as HTMLElement).getBoundingClientRect();
       const cardRect = (cardContainer.firstChild as HTMLElement).getBoundingClientRect();

       // Allow 5px tolerance for browser rendering differences
       expect(Math.abs(skeletonRect.height - cardRect.height)).toBeLessThan(5);
     });

     it('Skeleton has matching padding and border radius', () => {
       const { container } = render(<GameCardSkeleton />);
       const skeleton = container.firstChild as HTMLElement;
       const styles = window.getComputedStyle(skeleton);

       expect(styles.padding).toBe('24px'); // p-6 = 1.5rem = 24px
       expect(styles.borderRadius).toContain('8px'); // rounded-lg
     });
   });
   ```

Why getBoundingClientRect: Measures actual rendered dimensions, not just CSS classes
Why 5px tolerance: Browser rendering can vary slightly, strict equality is brittle
Why test badges separately: Status badge minimum width is critical for CLS prevention

   ```tsx
   import { render, fireEvent, screen } from '@testing-library/react';
   import { PullToRefresh } from '@/components/pull-to-refresh';

   describe('PullToRefresh', () => {
     it('Renders children', () => {
       render(
         <PullToRefresh onRefresh={() => {}}>
           <div>Test Content</div>
         </PullToRefresh>
       );

       expect(screen.getByText('Test Content')).toBeInTheDocument();
     });

     it('Triggers refresh when pull exceeds threshold', () => {
       const onRefresh = jest.fn();
       const { container } = render(
         <PullToRefresh onRefresh={onRefresh}>
           <div>Test Content</div>
         </PullToRefresh>
       );

       Object.defineProperty(window, 'scrollY', { value: 0, writable: true });

       const element = container.firstChild as HTMLElement;

       // Simulate pull gesture
       fireEvent.touchStart(element, { touches: [{ clientY: 100 }] });
       fireEvent.touchMove(element, { touches: [{ clientY: 200 }] }); // 100px pull
       fireEvent.touchEnd(element);

       expect(onRefresh).toHaveBeenCalled();
     });

     it('Does not trigger refresh below threshold', () => {
       const onRefresh = jest.fn();
       const { container } = render(
         <PullToRefresh onRefresh={onRefresh}>
           <div>Test Content</div>
         </PullToRefresh>
       );

       Object.defineProperty(window, 'scrollY', { value: 0, writable: true });

       const element = container.firstChild as HTMLElement;

       // Simulate small pull
       fireEvent.touchStart(element, { touches: [{ clientY: 100 }] });
       fireEvent.touchMove(element, { touches: [{ clientY: 150 }] }); // 50px pull (below 80px threshold)
       fireEvent.touchEnd(element);

       expect(onRefresh).not.toHaveBeenCalled();
     });
   });
   ```

Why mock APIs: Navigator APIs not available in jsdom, mocking simulates browser support
Why test cleanup: Ensures no memory leaks from event listeners
Why threshold testing: Verifies exact 80px pull requirement for UX consistency
